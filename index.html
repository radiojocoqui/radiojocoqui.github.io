function formatTime(seconds) {

  if (!Number.isFinite(seconds)) return "0:00";


  const minutes = Math.floor(seconds / 60);

  const remaining = Math.floor(seconds % 60);


  return minutes + ":" +

    String(remaining).padStart(2, "0");

}


function renderPlaylist() {

  playlist.innerHTML = "";


  tracks.forEach((track, index) => {

    const button = document.createElement("button");


    button.type = "button";

    button.className =

      index === currentTrack ? "track active" : "track";


    const name = document.createElement("span");

    name.textContent = track.title;


    const number = document.createElement("span");

    number.className = "track-number";

    number.textContent =

      String(index + 1).padStart(2, "0");


    button.append(name, number);


    button.addEventListener("click", () => {

      loadTrack(index, true);

    });


    playlist.appendChild(button);

  });

}


function updatePlayButton() {

  const playing = !audio.paused && !audio.ended;


  playButton.textContent = playing ? "❚❚" : "▶";

  playButton.setAttribute(

    "aria-label",

    playing ? "Pausar" : "Reproducir"

  );

}


function loadTrack(index, autoplay = false) {

  currentTrack =

    (index + tracks.length) % tracks.length;


  audio.pause();

  audio.src = tracks[currentTrack].file;

  audio.load();


  trackTitle.textContent = tracks[currentTrack].title;

  progress.value = 0;

  currentTimeLabel.textContent = "0:00";

  durationLabel.textContent = "0:00";

  statusLabel.textContent = "Lista para reproducir";


  renderPlaylist();

  updatePlayButton();


  if (autoplay) playCurrent();

}


async function playCurrent() {

  try {

    statusLabel.textContent = "Cargando audio...";

    await audio.play();

    statusLabel.textContent = "Transmitiendo música";

  } catch (error) {

    statusLabel.textContent =

      "Toca Play o verifica el archivo MP3";

  }


  updatePlayButton();

}

<script>

(function () {

  const holder = document.getElementById("sunTexture");

  const canvas = document.createElement("canvas");

  const ctx = canvas.getContext("2d");


  if (!ctx) return;


  canvas.width = 512;

  canvas.height = 512;

  holder.appendChild(canvas);


  const size = 512;

  const center = size / 2;

  const radius = 250;


  function noise(x, y, t) {

    return (

      Math.sin(x * 0.042 + t) *

      Math.cos(y * 0.055 - t * 0.6) +

      Math.sin((x + y) * 0.065 - t * 0.8) * 0.5 +

      Math.cos((x - y) * 0.09 + t * 1.3) * 0.25

    );

  }


  function drawSun(time) {

    const image = ctx.createImageData(size, size);

    const data = image.data;


    const t = time * 0.00013;


    for (let y = 0; y < size; y++) {

      for (let x = 0; x < size; x++) {

        const dx = x - center;

        const dy = y - center;

        const dist = Math.sqrt(dx * dx + dy * dy);


        const i = (y * size + x) * 4;


        if (dist > radius) {

          data[i + 3] = 0;

          continue;

        }


        const n = noise(x, y, t);

        const n2 = noise(x * 1.8, y * 1.8, -t * 1.5);


        const turbulence = n * 0.65 + n2 * 0.35;

        const edge = Math.pow(dist / radius, 1.7);


        let heat = 0.62 + turbulence * 0.15 - edge * 0.2;

        heat = Math.max(0, Math.min(1, heat));


        const spot = turbulence < -0.7 ? 0.5 : 1;


        data[i] = Math.min(255, 245 * spot);

        data[i + 1] = Math.min(

          255, (65 + heat * 165) * spot

        );

        data[i + 2] = Math.min(

          255, (5 + heat * 70) * spot

        );

        data[i + 3] = 225;

      }

    }


    ctx.putImageData(image, 0, 0);

  }


  const reducedMotion = window.matchMedia(

    "(prefers-reduced-motion: reduce)"

  ).matches;


  let last = 0;


  function animate(time) {

    if (time - last >= 100) {

      last = time;

      if (document.visibilityState === "visible") {

        drawSun(time);

      }

    }


    requestAnimationFrame(animate);

  }


  drawSun(0);


  if (!reducedMotion) {

    requestAnimationFrame(animate);

  }

})();

</script>

function makeTexture(planet) {

  const canvas = document.createElement("canvas");

  canvas.width = 512;

  canvas.height = 256;


  const ctx = canvas.getContext("2d");

  const w = canvas.width;

  const h = canvas.height;


  ctx.fillStyle = planet.base;

  ctx.fillRect(0, 0, w, h);


  const base = new THREE.Color(planet.base);

  const detail = new THREE.Color(planet.detail);


  for (let i = 0; i < 2200; i++) {

    const x = Math.random() * w;

    const y = Math.random() * h;


    let size = Math.random() * 18 + 2;


    if (planet.type === "gas" ||

        planet.type === "cloud") {

      size = Math.random() * 35 + 10;

    }


    const mix = Math.random();

    const color = base.clone().lerp(detail, mix);


    ctx.fillStyle = "#" + color.getHexString();

    ctx.globalAlpha = 0.15 + Math.random() * 0.45;


    ctx.beginPath();


    if (planet.type === "gas" ||

        planet.type === "cloud") {

      ctx.ellipse(

        x, y, size * 2, size * 0.18,

        0, 0, Math.PI * 2

      );

    } else {

      ctx.arc(x, y, size, 0, Math.PI * 2);

    }


    ctx.fill();

  }


  ctx.globalAlpha = 1;


  if (planet.type === "earth") {

    for (let i = 0; i < 70; i++) {

      const x = Math.random() * w;

      const y = Math.random() * h;


      ctx.fillStyle = "rgba(235,245,255,0.22)";

      ctx.beginPath();

      ctx.ellipse(

        x, y,

        15 + Math.random() * 30,

        2 + Math.random() * 5,

        Math.random() * 0.4,

        0, Math.PI * 2

      );

      ctx.fill();

    }

  }


  const texture = new THREE.CanvasTexture(canvas);

  texture.colorSpace = THREE.SRGBColorSpace;

  return texture;

}


function createPlanet(config) {

  const element = document.querySelector(config.selector);

  if (!element) return;


  let renderer;


  try {

    renderer = new THREE.WebGLRenderer({

      alpha: true,

      antialias: true,

      powerPreference: "low-power"

    });

  } catch (error) {

    console.warn("WebGL no disponible", error);

    return;

  }


  const width = element.clientWidth;

  const height = element.clientHeight;


  renderer.setPixelRatio(

    Math.min(window.devicePixelRatio || 1, 2)

  );


  renderer.setSize(width, height);

  renderer.setClearColor(0x000000, 0);


  const scene = new THREE.Scene();


  const camera = new THREE.PerspectiveCamera(

    40, width / height, 0.1, 100

  );


  camera.position.z = 3.8;


  const geometry = new THREE.SphereGeometry(

    1, 64, 48

  );


  const material = new THREE.MeshStandardMaterial({

    map: makeTexture(config),

    roughness: 0.9,

    metalness: 0

  });


  const sphere = new THREE.Mesh(geometry, material);

  sphere.rotation.z = 0.15;

  scene.add(sphere);


  scene.add(

    new THREE.AmbientLight(0xffffff, 0.65)

  );


  const sunlight = new THREE.DirectionalLight(

    0xffe2bd, 2.5

  );

  sunlight.position.set(-3, 3, 5);

  scene.add(sunlight);


  if (config.type === "gas") {

    const ringGeometry = new THREE.RingGeometry(

      1.25, 1.9, 96

    );


    const ringMaterial = new THREE.MeshBasicMaterial({

      color: 0xd9bd88,

      side: THREE.DoubleSide,

      transparent: true,

      opacity: 0.8,

      depthWrite: false

    });


    const rings = new THREE.Mesh(

      ringGeometry, ringMaterial

    );


    rings.rotation.x = 1.1;

    rings.rotation.z = -0.35;

    sphere.add(rings);


    camera.position.z = 5.5;

  }


  element.style.background = "transparent";

  element.style.boxShadow = "none";

  element.appendChild(renderer.domElement);


  instances.push({

    renderer, scene, camera, sphere,

    speed: config.speed

  });

}


planets.forEach(createPlanet);


const reducedMotion = window.matchMedia(

  "(prefers-reduced-motion: reduce)"

).matches;


let lastTime = 0;


function animate(time) {

  requestAnimationFrame(animate);


  if (time - lastTime < 33) return;

  lastTime = time;


  if (document.visibilityState !== "visible") return;


  instances.forEach(item => {

    if (!reducedMotion) {

      item.sphere.rotation.y += item.speed;

    }


    item.renderer.render(item.scene, item.camera);

  });

}


if (instances.length) {

  requestAnimationFrame(animate);

}
